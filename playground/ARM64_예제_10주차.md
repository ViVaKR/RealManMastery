# ARM64 예제 코드 — 10주차: 재귀 함수와 스택 프레임 심화

> 9주차에서 "LR 저장은 `bl`을 쓰는 함수라면 필수"라고 배웠는데, **재귀**는 이 원칙이
> 왜 "선택이 아니라 생명줄"이 되는지를 극단적으로 보여주는 사례네. 자기 자신을 부르는 순간,
> **매 호출마다 X30(LR)이 새 값으로 덮어써지므로, 이전 레벨의 복귀 주소를 각자 자기 스택 프레임에 저장해둬야만** 정상적으로 돌아올 수 있음.

---

## 예제 1 — 팩토리얼 (기본 재귀 패턴)

**C 등가 코드**
```c
int factorial(int n) {
    if (n <= 1) return 1;         // 종료 조건 (base case)
    return n * factorial(n - 1);  // 재귀 호출
}

extern void go_print_int(int v);

int main(void) {
    go_print_int(factorial(5));   // 120
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .text

// int factorial(int n)
    .align  2
factorial:
    cmp     w0, #1
    b.le    fact_base             // n <= 1 이면 종료 조건

    stp     x29, x30, [sp, #-32]!  // ★ 재귀 호출 직전이라 LR 저장이 필수!
    mov     x29, sp
    str     w0, [sp, #16]           // 이 레벨(depth)의 n을 이 레벨의 스택에 저장
                                     // ※ 매 재귀 호출마다 새 프레임이 생기므로, 각 레벨은 "자기만의 n 사본"을 가짐

    sub     w0, w0, #1               // n - 1
    bl      factorial                 // factorial(n-1) 재귀 호출 → w0에 결과

    ldr     w1, [sp, #16]             // 이 레벨의 n을 다시 로드
    mul     w0, w0, w1                 // n * factorial(n-1)

    ldp     x29, x30, [sp], #32       // 이 레벨의 LR 복원
    ret                                 // 이 레벨을 호출한 곳(한 단계 위 재귀 or main)으로 복귀

fact_base:
    mov     w0, #1
    ret                                 // leaf 상태(재귀 안 함)라 LR 저장 불필요

    .global _main
    .extern _go_print_int
_main:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    mov     w0, #5
    bl      factorial
    bl      _go_print_int             // 120

    mov     w0, #0
    ldp     x29, x30, [sp], #16
    ret
```

**포인트 — "각 재귀 레벨은 완전히 독립된 자기만의 스택 프레임을 가진다"**
`factorial(5)`를 호출하면 내부적으로 `factorial(4)`, `factorial(3)`... 이 차례로 `bl`되면서, **스택에 5개의 서로 다른 프레임이 쌓임**. 각 프레임은 자기 몫의 `n`(offset+16)과 자기 몫의 리턴주소(LR)를 따로 갖고 있어서 서로 섞이지 않네. 다음 그림으로 이해하면 쉬움:

```
[낮은 주소]
  ┌─────────────────┐ ← factorial(1) 프레임 (n=1, base case라 실제론 프레임 없음)
  ├─────────────────┤ ← factorial(2) 프레임 (n=2)
  ├─────────────────┤ ← factorial(3) 프레임 (n=3)
  ├─────────────────┤ ← factorial(4) 프레임 (n=4)
  ├─────────────────┤ ← factorial(5) 프레임 (n=5)
  ├─────────────────┤ ← main 프레임
[높은 주소]              (스택은 높은 주소 → 낮은 주소로 자람)
```

## 예제 2 — 피보나치 (재귀 호출이 2개일 때)

**C 등가 코드**
```c
int fib(int n) {
    if (n < 2) return n;
    return fib(n - 1) + fib(n - 2);   // 재귀 호출이 "두 번"!
}
```

**ARM64 어셈블리**
```asm
    .align  2
fib:
    cmp     w0, #2
    b.lt    fib_base                // n < 2 이면 n 그대로 리턴

    stp     x29, x30, [sp, #-32]!
    mov     x29, sp
    str     w0, [sp, #16]            // n 보관

    sub     w0, w0, #1
    bl      fib                       // fib(n-1) 계산
    str     w0, [sp, #20]             // ★ 결과를 스택에 임시 저장!
                                       // → 다음 bl fib(n-2) 호출이 w0를 또 덮어쓸 것이므로

    ldr     w0, [sp, #16]             // n 다시 로드
    sub     w0, w0, #2
    bl      fib                        // fib(n-2) 계산

    ldr     w1, [sp, #20]              // 아까 저장해둔 fib(n-1) 결과 회수
    add     w0, w0, w1                  // fib(n-1) + fib(n-2)

    ldp     x29, x30, [sp], #32
    ret

fib_base:
    ret                                  // n < 2 이면 n(w0) 그대로 리턴

    .global _main
    .extern _go_print_int
_main:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    mov     w0, #10
    bl      fib
    bl      _go_print_int              // 55

    mov     w0, #0
    ldp     x29, x30, [sp], #16
    ret
```

**포인트 — 왜 fib(n-1)의 결과를 스택에 잠깐 저장해야 하나**
9주차에서 배운 caller-saved 규칙이 그대로 적용되는 것이네: `fib(n-1)`의 결과(`w0`)를 들고 있는 상태에서 `bl fib`(n-2 계산)를 또 호출하면, **그 함수 내부에서 w0가 자유롭게 덮어써짐**. 그래서 첫 번째 결과를 **스택(또는 callee-saved 레지스터)에 잠깐 피신**시켜뒀다가, 두 번째 호출이 끝난 뒤 다시 꺼내 더해야 하네. 재귀가 2갈래로 갈라질 때마다 이 패턴이 반복됨.

## 예제 3 — 스택이 실제로 어떻게 쌓이는지 "눈으로" 확인하기

**C 등가 코드**
```c
extern void go_print_int(int v);

void print_descent(int n) {  // 내려가면서(재귀 호출 전에) 출력
    if (n == 0) return;
    go_print_int(n);
    print_descent(n - 1);
}

void print_ascent(int n) {   // 올라오면서(재귀 호출 후에) 출력
    if (n == 0) return;
    print_ascent(n - 1);
    go_print_int(n);
}
```

**ARM64 어셈블리**
```asm
    .align  2
print_descent:
    cbz     w0, desc_end              // ★ CBZ: "w0가 0이면 분기" — CMP+B.EQ를 한 방에! (이번 주 첫 등장)

    stp     x29, x30, [sp, #-32]!
    mov     x29, sp
    str     w0, [sp, #16]

    bl      _go_print_int              // ① 먼저 출력 (내려가는 중)
    ldr     w0, [sp, #16]
    sub     w0, w0, #1
    bl      print_descent               // ② 그다음 재귀

    ldp     x29, x30, [sp], #32
desc_end:
    ret

    .align  2
print_ascent:
    cbz     w0, asc_end

    stp     x29, x30, [sp, #-32]!
    mov     x29, sp
    str     w0, [sp, #16]

    sub     w0, w0, #1
    bl      print_ascent                 // ① 먼저 재귀 (끝까지 내려감)
    ldr     w0, [sp, #16]
    bl      _go_print_int                 // ② 그다음 출력 (돌아오는 중!)

    ldp     x29, x30, [sp], #32
asc_end:
    ret

    .global _main
    .extern _go_print_int
_main:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    mov     w0, #5
    bl      print_descent               // 출력: 5 4 3 2 1
    mov     w0, #5
    bl      print_ascent                // 출력: 1 2 3 4 5

    mov     w0, #0
    ldp     x29, x30, [sp], #16
    ret
```

**포인트 — 재귀 호출 "전"이냐 "후"냐가 순서를 완전히 뒤집는다**
- `print_descent`: 출력을 재귀 호출 **전**에 함 → 스택을 쌓아 내려가면서 바로바로 출력 → `5 4 3 2 1`
- `print_ascent`: 출력을 재귀 호출 **후**에 함 → base case까지 다 내려간 다음, **스택을 하나씩 걷어(pop)올라오면서** 출력 → `1 2 3 4 5`

`print_ascent`의 `go_print_int(n)`이 실행되는 시점은, **자기보다 더 깊은 재귀 호출들이 전부 끝나고 `ret`으로 돌아온 뒤**라는 걸 코드에서 직접 확인할 수 있네 — 이게 바로 "스택은 LIFO(후입선출)"라는 추상적 설명이 실제 코드에서 어떻게 나타나는지를 보여주는 예제일세.

**보너스 — `CBZ`/`CBNZ`**: `Wn`이 0인지/아닌지만 볼 때는 `CMP w0, #0` + `B.EQ`보다 **`CBZ w0, label`** 한 줄이 더 간결하네. 반대는 `CBNZ`(0이 아니면 분기). 플래그를 세팅하지 않는다는 차이도 있음(NZCV 안 건드림) — 재귀 종료조건처럼 단순 0 체크에 자주 쓰이는 관용구.

## 예제 4 — 스택 오버플로우, 그리고 컴파일러가 안 해주는 것

재귀가 너무 깊어지면 어떻게 될까?

```c
int infinite_recursion(int n) {
    return infinite_recursion(n + 1);   // 종료조건이 없음!
}
```

이 코드를 그대로 어셈블리로 옮기면, **매 호출마다 새 스택 프레임(이번 예제 기준 32바이트)이 계속 쌓이다가**, 결국 OS가 그 스레드에 할당한 스택 공간(예: macOS 메인 스레드 기본 8MB)을 다 써버리고 **세그멘테이션 폴트(정확히는 스택 오버플로우 시그널)**로 프로세스가 죽네.

**대략적인 계산**: 프레임 32바이트 기준, 8MB ÷ 32바이트 ≈ **약 26만 번**의 재귀 깊이에서 죽는다는 뜻 — `factorial(300000)`처럼 무모하게 깊은 재귀를 짜면 실제로 이 한계에 부딪힐 수 있음.

**⚠️ 흔한 오해 — "꼬리 재귀는 자동으로 최적화되지 않나요?"**
`return n * factorial(n-1)` 같은 형태(재귀 호출 뒤에 추가 연산이 남음, 즉 non-tail call)는 애초에 꼬리 재귀가 아니라서 최적화 대상이 아니고, **꼬리 재귀(tail call, 재귀 호출이 함수의 "마지막 동작"인 경우)라 해도 ARM64 자체가 자동으로 반복문으로 바꿔주지는 않음**. `clang -O2`로 컴파일하면 컴파일러가 꼬리 재귀를 감지해서 **`bl`+`ret` 쌍을 `b`(무조건분기)로 대체**하는 최적화를 해주긴 하는데, 이건 컴파일러의 최적화 옵션이지 ARM64 아키텍처가 보장하는 게 아니네. 우리가 손으로 짤 땐 항상 "이 재귀, 스택이 감당할 깊이인가?"를 스스로 따져야 함.

## 10주차 정리

| 개념 | 핵심 |
|---|---|
| 재귀에서 LR 저장 | 선택 아님, **필수** — 안 하면 복귀 주소를 잃어버림 |
| 각 재귀 레벨 | 완전히 독립된 자기만의 스택 프레임(지역변수+LR) 보유 |
| 재귀 호출이 2개 이상 | 첫 결과를 caller-saved 소멸 전에 스택/callee-saved로 피신 필요 |
| 호출 전 vs 호출 후 출력 | 스택의 LIFO 특성이 실행 순서를 결정 (`descent` vs `ascent`) |
| `CBZ`/`CBNZ` | 0 체크 전용 단축 명령 (NZCV 안 건드림) |
| 재귀 깊이 한계 | 스택 크기 ÷ 프레임 크기 = 대략적인 한계 재귀 깊이 |
| 꼬리 재귀 최적화 | 아키텍처가 아니라 **컴파일러 옵션**의 몫 |
