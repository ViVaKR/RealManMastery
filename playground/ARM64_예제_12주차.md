# ARM64 예제 코드 — 12주차: 구조체와 복합 자료구조

> 11주차에서 배운 "고정 스케일(×4, ×8) 인덱싱"에, 이번 주는 **"필드마다 서로 다른 고정 오프셋"**이라는
> 개념 하나만 더 얹으면 구조체가 완성되네. 그리고 그 두 개념(스케일드 인덱스 + 필드 오프셋)을 합치면
> **구조체 배열**이 나오고, 포인터로 다음 원소를 가리키게 하면 **연결 리스트**가 되는 것이네 — 전부 같은 뿌리일세.

---

## 예제 1 — 구조체 기본: 필드 오프셋은 그냥 상수

**C 등가 코드**
```c
struct Student {
    int id;      // offset 0
    int score;   // offset 4
};

extern void go_print_int(int v);

int main(void) {
    struct Student s;
    s.id = 1;
    s.score = 95;

    go_print_int(s.id);
    go_print_int(s.score);
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .text
    .global _main
    .extern _go_print_int

_main:
    stp     x29, x30, [sp, #-32]!   // FP/LR(16) + Student 구조체(8) → 정렬 위해 32
    mov     x29, sp

    // struct Student s; s.id = 1; s.score = 95;
    mov     w0, #1
    str     w0, [sp, #16]            // s.id    → 구조체 시작(sp+16) + offset 0
    mov     w0, #95
    str     w0, [sp, #20]            // s.score → 구조체 시작(sp+16) + offset 4

    ldr     w0, [sp, #16]
    bl      _go_print_int             // 1
    ldr     w0, [sp, #20]
    bl      _go_print_int             // 95

    mov     w0, #0
    ldp     x29, x30, [sp], #32
    ret
```

**포인트 — "구조체"는 컴파일러 머릿속에만 있다**
어셈블리 레벨에서 `s.id`, `s.score`라는 이름은 존재하지 않고, **그냥 `[sp, #16]`, `[sp, #20]`이라는 메모리 주소**만 있을 뿐이네. C 컴파일러가 `struct Student` 정의를 보고 "id는 시작주소+0, score는 시작주소+4"라는 **오프셋 표**를 컴파일 타임에 미리 계산해서, `s.score`라는 코드를 만날 때마다 그 상수 오프셋으로 치환해주는 것 — 이게 구조체의 정체일세.

## 예제 2 — 구조체 배열 (11주차 스케일드 인덱스 + 필드 오프셋의 결합)

**C 등가 코드**
```c
struct Student { int id; int score; };   // sizeof(Student) == 8

extern void go_print_int(int v);

int main(void) {
    struct Student students[3] = { {1,95}, {2,82}, {3,67} };
    int sum = 0;

    for (int i = 0; i < 3; i++) {
        sum += students[i].score;
    }

    go_print_int(sum / 3);   // 81
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .text
    .global _main
    .extern _go_print_int

_main:
    stp     x29, x30, [sp, #-48]!    // FP/LR(16) + Student[3] (8*3=24 → 정렬위해 32) = 48
    mov     x29, sp

    mov     w0, #1
    str     w0, [sp, #16]             // students[0].id
    mov     w0, #95
    str     w0, [sp, #20]             // students[0].score
    mov     w0, #2
    str     w0, [sp, #24]             // students[1].id
    mov     w0, #82
    str     w0, [sp, #28]             // students[1].score
    mov     w0, #3
    str     w0, [sp, #32]             // students[2].id
    mov     w0, #67
    str     w0, [sp, #36]             // students[2].score

    add     x21, sp, #16               // students 배열 시작주소
    mov     w19, #0                     // sum
    mov     w20, #0                     // i

sum_loop:
    cmp     w20, #3
    b.ge    sum_end

    lsl     w1, w20, #3                 // i * sizeof(Student) = i * 8   ★ 스케일드 인덱스(11주차)
    add     x2, x21, w1, sxtw            // &students[i]
    ldr     w0, [x2, #4]                  // students[i].score          ★ 필드 오프셋(이번 주)

    add     w19, w19, w0
    add     w20, w20, #1
    b       sum_loop

sum_end:
    mov     w20, #3
    sdiv    w0, w19, w20
    bl      _go_print_int                  // 81 ((95+82+67)/3)

    mov     w0, #0
    ldp     x29, x30, [sp], #48
    ret
```

**포인트 — 구조체 배열 주소 계산은 2단계**
1. `&students[i]` = `base + i * sizeof(Student)` → **`LSL #3`**(구조체 크기 8바이트라서 ×8)
2. `students[i].score` = `&students[i] + 4` → **필드 오프셋은 그냥 즉치 덧셈**

**"어떤 원소인가"(스케일드 인덱스)**와 **"그 원소의 어떤 필드인가"(고정 오프셋)**를 완전히 분리해서 계산한다는 게 이번 예제의 핵심이네.

## 예제 3 — 연결 리스트 (포인터로 다음 원소 가리키기)

**C 등가 코드**
```c
struct Node {
    long value;         // offset 0
    struct Node* next;  // offset 8
};

extern void go_print_int(int v);

int main(void) {
    struct Node node0, node1, node2;

    node0.value = 10;  node0.next = &node1;
    node1.value = 20;  node1.next = &node2;
    node2.value = 30;  node2.next = NULL;

    struct Node* p = &node0;
    while (p != NULL) {
        go_print_int((int)p->value);
        p = p->next;
    }
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .text
    .global _main
    .extern _go_print_int

_main:
    stp     x29, x30, [sp, #-64]!    // FP/LR(16) + Node 3개(16바이트씩=48) = 64
    mov     x29, sp

    // node0 @ sp+16 (value: sp+16, next: sp+24)
    mov     x0, #10
    str     x0, [sp, #16]             // node0.value
    add     x1, sp, #32                // &node1
    str     x1, [sp, #24]              // node0.next = &node1

    // node1 @ sp+32 (value: sp+32, next: sp+40)
    mov     x0, #20
    str     x0, [sp, #32]
    add     x1, sp, #48                 // &node2
    str     x1, [sp, #40]               // node1.next = &node2

    // node2 @ sp+48 (value: sp+48, next: sp+56)
    mov     x0, #30
    str     x0, [sp, #48]
    mov     x1, #0                       // NULL
    str     x1, [sp, #56]                // node2.next = NULL

    // ── 순회: p = &node0; while (p) { print(p->value); p = p->next; } ──
    add     x21, sp, #16                  // struct Node* p = &node0

traverse_loop:
    cbz     x21, traverse_end              // p == NULL 이면 종료

    ldr     w0, [x21, #0]                   // p->value (offset 0)
    bl      _go_print_int                    // 10 → 20 → 30 순서로 출력

    ldr     x21, [x21, #8]                   // p = p->next (offset 8) ★ 포인터 자체를 갱신!

    b       traverse_loop

traverse_end:
    mov     w0, #0
    ldp     x29, x30, [sp], #64
    ret
```

**포인트 — 연결 리스트는 "포인터를 필드로 가진 구조체"일 뿐**
`Node.next`라는 필드가 **다음 노드의 주소를 값으로 가지고 있는 8바이트 슬롯**이라는 것 말고는, 예제 1의 구조체와 다를 게 없네. 순회 루프의 핵심은 `ldr x21, [x21, #8]` 단 한 줄 — **"현재 포인터가 가리키는 곳에서 next 필드를 읽어와, 그 값을 포인터 자신에 다시 대입"**하는 것이네. 배열은 인덱스를 증가시키며 순회하고(11주차), 연결 리스트는 포인터를 갱신하며 순회한다는 차이만 있을 뿐, **"메모리 어딘가를 계속 따라간다"**는 본질은 같음.

## 예제 4 — 문자열 순회 (NUL-terminated string)

**C 등가 코드**
```c
extern int  sys_print(const char* fmt);
extern void go_print_int(int v);

int my_strlen(const char* s) {
    int len = 0;
    while (s[len] != '\0') {
        len++;
    }
    return len;
}

int main(void) {
    const char* msg = "ARM64 화이팅!\n";
    sys_print(msg);
    go_print_int(my_strlen(msg));
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .data
    .align  3
msg: .asciz "ARM64 화이팅!\n"     // .asciz가 문자열 끝에 자동으로 '\0' 붙여줌

    .text
    .extern _sys_print
    .extern _go_print_int

// int my_strlen(const char* s)  — leaf function
    .align  2
my_strlen:
    mov     w1, #0                    // len = 0
strlen_loop:
    ldrb    w2, [x0, w1, sxtw]         // s[len] 한 바이트 로드   ★ 문자열은 원소크기 1바이트라 스케일 불필요
    cbz     w2, strlen_end              // '\0'(0)이면 종료
    add     w1, w1, #1                  // len++
    b       strlen_loop
strlen_end:
    mov     w0, w1                      // 리턴값 = len
    ret

    .global _main
_main:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    adrp    x0, msg@PAGE
    add     x0, x0, msg@PAGEOFF
    bl      _sys_print                   // "ARM64 화이팅!" 출력 (Rust 브릿지)

    adrp    x0, msg@PAGE
    add     x0, x0, msg@PAGEOFF
    bl      my_strlen
    bl      _go_print_int                 // 문자열 길이 출력

    mov     w0, #0
    ldp     x29, x30, [sp], #16
    ret
```

**포인트 — 문자열도 결국 "1바이트짜리 원소의 배열"**
`char` 배열(문자열)은 11주차의 배열 원리에서 **원소 크기가 1바이트라 스케일(`LSL`)이 필요 없는 특수 케이스**일 뿐이네. `LDRB`(byte load)로 한 글자씩 읽고, **끝을 알리는 별도의 "길이" 정보 없이 `'\0'`(0)이라는 특수값을 만나면 멈추는 방식**(NUL-terminated)이 C 문자열의 핵심 관례고, 이 관례를 그대로 어셈블리로 구현한 게 `strlen_loop`이네. `CBZ`(10주차에서 배운)가 "0바이트 만나면 종료"를 처리하는 데 정확히 들어맞는다는 것도 확인할 수 있고.

## 12주차 정리 — 이번 주 배운 모든 게 사실 하나였다

| 자료구조 | 주소 계산 공식 | 이번 주 어디서 |
|---|---|---|
| 구조체 필드 | `base + 고정오프셋` | 예제 1 |
| 구조체 배열 원소 | `base + i * sizeof(구조체)` | 예제 2 (앞부분) |
| 구조체 배열의 필드 | `(base + i*sizeof) + 고정오프셋` | 예제 2 (전체) |
| 연결 리스트 | `현재포인터 + next오프셋` → 그 값을 포인터에 대입 | 예제 3 |
| 문자열(=1바이트 배열) | `base + i * 1` (스케일 불필요) | 예제 4 |

11주차의 **"배열 인덱싱 = 주소 계산"**과 12주차의 **"구조체 필드 = 고정 오프셋"**을 합치면 사실상 **모든 복합 자료구조(구조체, 배열, 문자열, 연결 리스트)가 전부 "base + offset" 하나의 공식으로 환원**된다는 게 이번 두 주차를 관통하는 결론이네.
