# ARM64 예제 코드 — 4주차: 산술/논리 연산자

> 계속해서 `_main` 진입점 + 브릿지 라이브러리(`go_print_int`, `go_add` 등) 출력 방식으로 진행하네.
> 이번 주는 결과 출력을 주로 Go의 `go_print_int(int32 v)`로 통일해서, 연산 자체에 집중하도록 구성했네.

---

## 예제 1 — 사칙연산 (ADD / SUB / MUL / SDIV)

**C 등가 코드**
```c
extern void go_print_int(int v);

int main(void) {
    int a = 17, b = 5;
    int sum = a + b;   // 22
    int sub = a - b;   // 12
    int mul = a * b;   // 85
    int div = a / b;   // 3 (정수 나눗셈, 몫만)

    go_print_int(sum);
    go_print_int(sub);
    go_print_int(mul);
    go_print_int(div);
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .text
    .global _main
    .extern _go_print_int

_main:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    mov     w19, #17            // a  (※ w19는 callee-saved, 여러 연산에 재사용하려고 보관용으로 사용)
    mov     w20, #5             // b

    add     w0, w19, w20        // sum = a + b
    bl      _go_print_int

    sub     w0, w19, w20        // sub = a - b
    bl      _go_print_int

    mul     w0, w19, w20        // mul = a * b
    bl      _go_print_int

    sdiv    w0, w19, w20        // div = a / b (부호있는 나눗셈)
    bl      _go_print_int

    mov     w0, #0
    ldp     x29, x30, [sp], #16
    ret
```

**⚠️ 주의 — callee-saved 레지스터를 쓴 이유**
`w19`는 이번 주 예제에서 처음 등장하는데, `bl`로 다른 함수를 호출하면 **caller-saved 레지스터(X0~X18)는 값이 보존된다는 보장이 없음**. `a`, `b`를 여러 번의 `bl` 호출 사이에서 계속 써야 하니 **callee-saved 레지스터(X19~X28)**에 담아둔 것이네. (X19~X28은 9~10주차 함수 호출 규약에서 본격적으로 다룰 텐데, 여기서 살짝 맛보기로 필요해서 미리 씀 — 지금은 "값을 여러 함수 호출 너머로 보존하려면 X19~X28을 쓴다" 정도만 기억해두게)

**포인트**
| C 연산자 | ARM64 명령 | 비고 |
|---|---|---|
| `+` | `ADD` | |
| `-` | `SUB` | |
| `*` | `MUL` | |
| `/` (정수, 부호있음) | `SDIV` | 몫만 계산, 나머지는 안 나옴 |
| `/` (정수, 부호없음) | `UDIV` | unsigned 버전 |

## 예제 2 — 나머지 연산 (MOD 패턴)

ARM64에는 `MOD` 명령이 **없음**. 대신 "나눈 몫에 다시 나눈 값을 곱해서 원래 값에서 빼는" 패턴을 쓰네: `a % b == a - (a / b) * b`

**C 등가 코드**
```c
extern void go_print_int(int v);

int main(void) {
    int a = 17, b = 5;
    int mod = a % b;   // 2
    go_print_int(mod);
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .text
    .global _main
    .extern _go_print_int

_main:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    mov     w0, #17             // a
    mov     w1, #5              // b

    sdiv    w2, w0, w1          // w2 = a / b  (몫, 3)
    msub    w3, w2, w1, w0      // w3 = a - (w2 * w1) = 17 - 15 = 2  (MSUB: 곱셈 후 뺄셈을 한 명령으로!)

    mov     w0, w3
    bl      _go_print_int

    mov     w0, #0
    ldp     x29, x30, [sp], #16
    ret
```

**포인트**
- `MSUB Wd, Wn, Wm, Wa` = `Wd = Wa - (Wn * Wm)` — 곱셈+뺄셈을 명령 하나로 처리하는 **곱셈-누산(multiply-accumulate)** 계열 명령
- 컴파일러도 `%` 연산을 실제로 이렇게 `SDIV + MSUB` 조합으로 풀어냄 (objdump로 확인해보면 똑같이 나올 걸세)

## 예제 3 — 논리 연산 (AND / ORR / EOR / MVN)과 비트마스크

**C 등가 코드**
```c
extern void go_print_int(int v);

int main(void) {
    unsigned int flags = 0b1010;      // 10
    unsigned int mask  = 0b0110;      // 6

    unsigned int r_and = flags & mask;   // 0010 = 2
    unsigned int r_or  = flags | mask;   // 1110 = 14
    unsigned int r_xor = flags ^ mask;   // 1100 = 12
    unsigned int r_not = ~flags;         // 비트 반전

    go_print_int(r_and);
    go_print_int(r_or);
    go_print_int(r_xor);
    go_print_int(r_not);
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .text
    .global _main
    .extern _go_print_int

_main:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    mov     w19, #0b1010        // flags = 10
    mov     w20, #0b0110        // mask  = 6

    and     w0, w19, w20        // AND
    bl      _go_print_int

    orr     w0, w19, w20        // OR
    bl      _go_print_int

    eor     w0, w19, w20        // XOR
    bl      _go_print_int

    mvn     w0, w19             // NOT (비트 반전, ~flags)
    bl      _go_print_int

    mov     w0, #0
    ldp     x29, x30, [sp], #16
    ret
```

**포인트**
| C 연산자 | ARM64 명령 |
|---|---|
| `&` | `AND` |
| `\|` | `ORR` |
| `^` | `EOR` |
| `~` | `MVN` (move-not: `Wd = ~Wn`) |

`~flags` 같은 단항 연산자에는 대응 명령이 `NOT`이 아니라 `MVN`(**M**o**V**e **N**ot)이라는 이름인 게 처음엔 낯설 수 있는데, "값을 반전해서 옮긴다"는 뜻으로 이해하면 되네.

## 예제 4 — 시프트 연산 (LSL / LSR / ASR / ROR)

**C 등가 코드**
```c
extern void go_print_int(int v);

int main(void) {
    int a = -16;
    unsigned int b = 16;

    int left      = b << 2;   // 64  (논리 좌시프트)
    unsigned int r_logic = b >> 2;   // 4   (논리 우시프트, 부호없음)
    int r_arith   = a >> 2;   // -4  (산술 우시프트, 부호 유지!)

    go_print_int(left);
    go_print_int(r_logic);
    go_print_int(r_arith);
    return 0;
}
```

**ARM64 어셈블리**
```asm
    .text
    .global _main
    .extern _go_print_int

_main:
    stp     x29, x30, [sp, #-16]!
    mov     x29, sp

    mov     w19, #16            // b (unsigned 취급)
    mov     w20, #-16           // a (signed)

    lsl     w0, w19, #2         // 논리 좌시프트: b << 2 = 64
    bl      _go_print_int

    lsr     w0, w19, #2         // 논리 우시프트: b >> 2 = 4  (빈 자리는 0으로 채움)
    bl      _go_print_int

    asr     w0, w20, #2         // 산술 우시프트: a >> 2 = -4 (빈 자리를 부호비트로 채움!)
    bl      _go_print_int

    mov     w0, #0
    ldp     x29, x30, [sp], #16
    ret
```

**포인트 — 왜 우시프트가 2종류인가**
| C 연산자 | ARM64 명령 | 빈 자리를 채우는 값 | 용도 |
|---|---|---|---|
| `<<` | `LSL` | 0 (오른쪽에서 채움) | 부호 상관없이 동일 |
| `>>` (unsigned) | `LSR` | 0 | 논리 시프트 |
| `>>` (signed) | `ASR` | 부호비트(MSB) | 음수의 부호를 보존해야 하므로 |
| (C엔 없음, 회전) | `ROR` | 반대쪽 끝 비트가 다시 들어옴 | 해시/암호 연산 등에서 사용 |

**컴파일러 관점 팁**: C에서 `int`(signed) 우시프트는 `ASR`로, `unsigned int` 우시프트는 `LSR`로 컴파일됨 — **같은 `>>` 연산자라도 자료형에 따라 다른 명령어가 나온다**는 걸 여기서 확인할 수 있네. 2주차에서 배운 "자료형 = 해석 방식"이 여기서도 그대로 이어지는 셈이지.

## 4주차 정리 — C 연산자 ↔ ARM64 명령 총정리표

| 분류 | C 연산자 | ARM64 명령 |
|---|---|---|
| 산술 | `+` `-` `*` | `ADD` `SUB` `MUL` |
| 나눗셈 | `/` (signed/unsigned) | `SDIV` / `UDIV` |
| 나머지 | `%` | `SDIV` + `MSUB` 조합 |
| 논리 | `&` `\|` `^` `~` | `AND` `ORR` `EOR` `MVN` |
| 시프트 | `<<` | `LSL` |
| 시프트 | `>>` (unsigned) | `LSR` |
| 시프트 | `>>` (signed) | `ASR` |
| (회전) | 없음 | `ROR` |
| (곱셈+누산) | 없음 (컴파일러 최적화용) | `MADD` / `MSUB` |
